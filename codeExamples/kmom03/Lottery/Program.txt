Console.WriteLine($"Mata in {LotteryLine.NR_TO_MARK} tal inom [{LotteryLine.MIN_NUMBER}, {LotteryLine.MAX_NUMBER}]: ");
while (!lottLine.IsFullyMarked())
{
    
    string input = Console.ReadLine() ?? "-1";
    if (!int.TryParse(input, out int nr))
    {
        Console.WriteLine("Du måste mata in ett heltal");
    }
    lottLine.MarkNr(nr);
    Console.WriteLine($"Antalet kryssade: {lottLine.GetNrOfMarkedNumbers()}");
}
